import {
  handleGetProfile,
  handleUpdateName,
  handleUpdateEmail,
  handleUpdateBI,
  handleUpdatePhone,
  handleUpdatePassword,
  handleDeleteAccount,
} from "@/api/userServices";
import { Loader } from "@/components/Loader";
import NoContent from "@/components/NoContent";
import { useAuth } from "@/contexts/AuthContext";
import type { IUser } from "@/types/auth";
import { useEffect, useState } from "react";

export default function Profile() {
  const [user, setUser] = useState<IUser | null>(null);
  const [loading, setLoading] = useState<boolean>(false);
  const [updateLoading, setUpdateLoading] = useState<string>("");
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState<string | null>(null);
  const { login } = useAuth();
  const [formData, setFormData] = useState({
    name: "",
    email: "",
    bi: "",
    phone: "",
    password: "",
    newPassword: "",
  });

  useEffect(() => {
    getProfile();
  }, []);

  const getProfile = async () => {
    setLoading(true);
    try {
      const res = await handleGetProfile();
      setUser(res.data);
      setFormData({
        name: res.data.name || "",
        email: res.data.email || "",
        bi: res.data.bi || "",
        phone: res.data.phoneNumber || "",
        password: "",
        newPassword: "",
      });
    } catch {
      setError("Erro ao carregar perfil");
    } finally {
      setLoading(false);
    }
  };

  const handleUpdate = async (
    fieldKey: string,
    updateFn: () => Promise<any>,
    updateAuth: boolean = false
  ) => {
    setUpdateLoading(fieldKey.toLowerCase());
    setError(null);
    setSuccess(null);

    try {
      const res = await updateFn();
      if (updateAuth) {
        login({ ...res.data, isAdmin: res.data.role === "ADMIN" });
      }
      setSuccess(`${fieldKey} atualizado com sucesso!`);
      await getProfile();
    } catch (err: any) {
      setError(err.response?.data?.message || `Erro ao atualizar ${fieldKey}`);
    } finally {
      setUpdateLoading("");
      setTimeout(() => setError(null), 3000);
      setTimeout(() => setSuccess(null), 3000);
    }
  };

  const UpdateField = ({
    label,
    fieldKey,
    type = "text",
    value,
    placeholder,
    onChange,
    onUpdate,
  }: {
    label: string;
    fieldKey: string;
    type?: string;
    value: string;
    placeholder: string;
    onChange: (e: React.ChangeEvent<HTMLInputElement>) => void;
    onUpdate: () => void;
  }) => (
    <div className="flex flex-col sm:flex-row items-start sm:items-center gap-4">
      <div className="w-full sm:w-1/3">
        <p className="font-semibold">{label}:</p>
        <p className="text-sm text-gray-600 break-words">{user?.[fieldKey as keyof IUser]}</p>
      </div>
      <div className="w-full sm:w-2/3 flex flex-col sm:flex-row gap-2">
        <input
          type={type}
          value={value}
          onChange={onChange}
          className="flex-1 p-2 border rounded"
          placeholder={placeholder}
        />
        <button
          onClick={onUpdate}
          disabled={updateLoading === fieldKey.toLowerCase()}
          className="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600 disabled:bg-gray-400"
        >
          {updateLoading === fieldKey.toLowerCase() ? "Atualizando..." : "Atualizar"}
        </button>
      </div>
    </div>
  );

  if (loading) return <Loader label="Carregando perfil..." />;
  if (!user)
    return (
      <NoContent
        title="Você não está autenticado"
        description="Inicie sessão para aceder ao perfil"
      />
    );

  return (
    <div className="min-h-screen p-6 max-w-4xl mx-auto">
      <h1 className="text-2xl font-bold mb-6">Meu Perfil</h1>

      {error && <Alert type="error" message={error} />}
      {success && <Alert type="success" message={success} />}

      <div className="space-y-6">
        <UpdateField
          label="Nome"
          fieldKey="name"
          value={formData.name}
          placeholder="Novo nome"
          onChange={(e) => setFormData({ ...formData, name: e.target.value })}
          onUpdate={() =>
            handleUpdate("Nome", () => handleUpdateName({ name: formData.name }))
          }
        />

        <UpdateField
          label="Email"
          fieldKey="email"
          type="email"
          value={formData.email}
          placeholder="Novo email"
          onChange={(e) => setFormData({ ...formData, email: e.target.value })}
          onUpdate={() =>
            handleUpdate(
              "Email",
              () => handleUpdateEmail({ email: formData.email, password: formData.password }),
              true
            )
          }
        />

        <UpdateField
          label="BI"
          fieldKey="bi"
          value={formData.bi}
          placeholder="Novo BI"
          onChange={(e) => setFormData({ ...formData, bi: e.target.value })}
          onUpdate={() => handleUpdate("BI", () => handleUpdateBI({ bi: formData.bi }))}
        />

        <UpdateField
          label="Telefone"
          fieldKey="phoneNumber"
          type="tel"
          value={formData.phone}
          placeholder="Novo telefone"
          onChange={(e) => setFormData({ ...formData, phone: e.target.value })}
          onUpdate={() =>
            handleUpdate("Telefone", () => handleUpdatePhone({ phoneNumber: formData.phone }))
          }
        />

        <div className="border-t pt-6">
          <h2 className="text-xl font-semibold mb-4">Alterar Senha</h2>
          <div className="space-y-4">
            <input
              type="password"
              value={formData.password}
              onChange={(e) =>
                setFormData({ ...formData, password: e.target.value })
              }
              className="w-full p-2 border rounded"
              placeholder="Senha atual"
            />
            <input
              type="password"
              value={formData.newPassword}
              onChange={(e) =>
                setFormData({ ...formData, newPassword: e.target.value })
              }
              className="w-full p-2 border rounded"
              placeholder="Nova senha"
            />
            <button
              onClick={() =>
                handleUpdate("senha", () =>
                  handleUpdatePassword({
                    password: formData.password,
                    newPassword: formData.newPassword,
                  })
                )
              }
              disabled={updateLoading === "senha"}
              className="w-full px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600 disabled:bg-gray-400"
            >
              {updateLoading === "senha" ? "Atualizando senha..." : "Atualizar senha"}
            </button>
          </div>
        </div>

        <div className="border-t pt-6">
          <button
            onClick={async () => {
              if (
                window.confirm("Tem certeza que deseja excluir sua conta? Esta ação não pode ser desfeita.")
              ) {
                try {
                  await handleDeleteAccount();
                  // Redirecionamento após exclusão
                } catch (err: any) {
                  setError(err.response?.data?.message || "Erro ao excluir conta");
                }
              }
            }}
            className="px-4 py-2 bg-red-500 text-white rounded hover:bg-red-600"
          >
            Excluir conta
          </button>
        </div>
      </div>
    </div>
  );
}

function Alert({ type, message }: { type: "error" | "success"; message: string }) {
  const styles = {
    error: "bg-red-100 border-red-400 text-red-700",
    success: "bg-green-100 border-green-400 text-green-700",
  };
  return (
    <div className={`border px-4 py-3 rounded mb-4 ${styles[type]}`}>
      {message}
    </div>
  );
}
